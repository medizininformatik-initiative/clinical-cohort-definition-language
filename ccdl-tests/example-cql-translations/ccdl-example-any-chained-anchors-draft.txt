  exists (from ((from [Condition: Code 'F05' from codeSystem1] C
          where ToDate(C.recordedDate as dateTime) in Interval["AnchorDate_anchor-dementia-diagnosis" + 0 hours, "AnchorDate_anchor-dementia-diagnosis" + 720 hours]
          return ToDate(C.recordedDate as dateTime)) union
        (from [Condition: Code 'F05.8' from codeSystem1] C
          where ToDate(C.recordedDate as dateTime) in Interval["AnchorDate_anchor-dementia-diagnosis" + 0 hours, "AnchorDate_anchor-dementia-diagnosis" + 720 hours]
          return ToDate(C.recordedDate as dateTime)) union

          where ToDate(C.recordedDate as dateTime) in Interval["AnchorDate_anchor-dementia-diagnosis" + 0 hours, "AnchorDate_anchor-dementia-diagnosis" + 720 hours]
          return ToDate(C.recordedDate as dateTime)) union
