library Retrieve version '1.0.0'
using FHIR version '4.0.0'
include FHIRHelpers version '4.0.0'

codesystem codeSystem1: 'http://fhir.de/CodeSystem/bfarm/ops'
codesystem codeSystem2: 'http://loinc.org'

context Patient

define "AnchorDate_anchor-procedure":
  Min((from [Procedure: Code '5-470.1' from codeSystem1] P
    return Coalesce(ToDate(P.performed as dateTime), ToDate((P.performed as Period).start))) union
  (from [Procedure: Code '5-470.1x' from codeSystem1] P
    return Coalesce(ToDate(P.performed as dateTime), ToDate((P.performed as Period).start))) union
  (from [Procedure: Code '5-470.10' from codeSystem1] P
    return Coalesce(ToDate(P.performed as dateTime), ToDate((P.performed as Period).start))) union
  (from [Procedure: Code '5-470.11' from codeSystem1] P
    return Coalesce(ToDate(P.performed as dateTime), ToDate((P.performed as Period).start))))

define Criterion:
  exists (from [Observation: Code '718-7' from codeSystem2] O
    where ToDate(O.effective as dateTime) in Interval["AnchorDate_anchor-procedure" + 0 hours, "AnchorDate_anchor-procedure" + 24 hours])

define InInitialPopulation:
  "AnchorDate_anchor-procedure" is not null and
  Criterion
