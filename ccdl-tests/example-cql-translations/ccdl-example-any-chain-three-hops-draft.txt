library Retrieve version '1.0.0'
using FHIR version '4.0.0'
include FHIRHelpers version '4.0.0'

codesystem codeSystem1: 'http://fhir.de/CodeSystem/bfarm/icd-10-gm'
codesystem codeSystem2: 'http://fhir.de/CodeSystem/bfarm/ops'
codesystem codeSystem3: 'http://loinc.org'

context Patient

define InInitialPopulation:
  exists (from ((from [Condition: Code 'A41.5' from codeSystem1] C
          return ToDate(C.recordedDate as dateTime)) union
        (from [Condition: Code 'A41.58' from codeSystem1] C
          return ToDate(C.recordedDate as dateTime)) union
        (from [Condition: Code 'A41.51' from codeSystem1] C
          return ToDate(C.recordedDate as dateTime)) union
        (from [Condition: Code 'A41.52' from codeSystem1] C
          return ToDate(C.recordedDate as dateTime))) W1
    where exists (from ((from [Condition: Code 'N17.0' from codeSystem1] C
              where ToDate(C.recordedDate as dateTime) in Interval[W1 + 0 hours, W1 + 168 hours]
              return ToDate(C.recordedDate as dateTime)) union
            (from [Condition: Code 'N17.09' from codeSystem1] C
              where ToDate(C.recordedDate as dateTime) in Interval[W1 + 0 hours, W1 + 168 hours]
              return ToDate(C.recordedDate as dateTime)) union
            (from [Condition: Code 'N17.02' from codeSystem1] C
              where ToDate(C.recordedDate as dateTime) in Interval[W1 + 0 hours, W1 + 168 hours]
              return ToDate(C.recordedDate as dateTime)) union
            (from [Condition: Code 'N17.03' from codeSystem1] C
              where ToDate(C.recordedDate as dateTime) in Interval[W1 + 0 hours, W1 + 168 hours]
              return ToDate(C.recordedDate as dateTime)) union
            (from [Condition: Code 'N17.01' from codeSystem1] C
              where ToDate(C.recordedDate as dateTime) in Interval[W1 + 0 hours, W1 + 168 hours]
              return ToDate(C.recordedDate as dateTime))) W2
        where exists (from ((from [Procedure: Code '8-85a.0' from codeSystem2] P
                  where ToDate(P.performed as dateTime) in Interval[W2 + 0 hours, W2 + 336 hours] or
                    P.performed overlaps Interval[W2 + 0 hours, W2 + 336 hours]
                  return Coalesce(ToDate(P.performed as dateTime), ToDate((P.performed as Period).start))) union
                (from [Procedure: Code '8-85a.02' from codeSystem2] P
                  where ToDate(P.performed as dateTime) in Interval[W2 + 0 hours, W2 + 336 hours] or
                    P.performed overlaps Interval[W2 + 0 hours, W2 + 336 hours]
                  return Coalesce(ToDate(P.performed as dateTime), ToDate((P.performed as Period).start))) union
                (from [Procedure: Code '8-85a.01' from codeSystem2] P
                  where ToDate(P.performed as dateTime) in Interval[W2 + 0 hours, W2 + 336 hours] or
                    P.performed overlaps Interval[W2 + 0 hours, W2 + 336 hours]
                  return Coalesce(ToDate(P.performed as dateTime), ToDate((P.performed as Period).start))) union
                (from [Procedure: Code '8-85a.03' from codeSystem2] P
                  where ToDate(P.performed as dateTime) in Interval[W2 + 0 hours, W2 + 336 hours] or
                    P.performed overlaps Interval[W2 + 0 hours, W2 + 336 hours]
                  return Coalesce(ToDate(P.performed as dateTime), ToDate((P.performed as Period).start))) union
                (from [Procedure: Code '8-85a.00' from codeSystem2] P
                  where ToDate(P.performed as dateTime) in Interval[W2 + 0 hours, W2 + 336 hours] or
                    P.performed overlaps Interval[W2 + 0 hours, W2 + 336 hours]
                  return Coalesce(ToDate(P.performed as dateTime), ToDate((P.performed as Period).start)))) W3
            where exists (from [Observation: Code '718-7' from codeSystem3] O
                where ToDate(O.effective as dateTime) in Interval[W3 + 0 hours, W3 + 24 hours]))))
